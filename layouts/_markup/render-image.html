{{- /* Render markdown images from assets/ so the browser gets intrinsic dimensions (no layout shift) and can defer offscreen loads. Falls back to a plain for anything that is not a local asset, e.g. remote URLs or static/ files. */ -}} {{- $src := .Destination -}} {{- $res := "" -}} {{- with .PageInner.Resources.GetMatch $src -}} {{- $res = . -}} {{- else -}} {{- $res = resources.Get $src -}} {{- end -}} {{- if $res -}} {{- else -}} {{- warnf "render-image: no resource found for %q in %s" $src .Page.File.Path -}} {{- end -}}
